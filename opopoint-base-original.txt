function diasRestantes(fechaISO) {
  const hoy = new Date();
  const fin = new Date(fechaISO + "T23:59:59");
  return Math.ceil((fin - hoy) / (1000 * 60 * 60 * 24));
}
function formatFecha(fechaISO) {
  const d = new Date(fechaISO + "T00:00:00");
  return d.toLocaleDateString("es-ES", { day: "2-digit", month: "short", year: "numeric" });
}

function OpoScope() {
  const [convocatorias, setConvocatorias] = useState(seedConvocatorias);
  const [query, setQuery] = useState("");
  const [provincia, setProvincia] = useState("Todas");
  const [grupo, setGrupo] = useState("Todos");
  const [toast, setToast] = useState(null);
  const [newIds, setNewIds] = useState(new Set());
  const [lastSync, setLastSync] = useState(new Date());
  const [syncing, setSyncing] = useState(false);
  const poolIndexRef = useRef(0);

  const filtered = useMemo(() => {
    const q = query.trim().toLowerCase();
    return convocatorias
      .filter((c) => provincia === "Todas" || c.provincia === provincia)
      .filter((c) => grupo === "Todos" || c.grupo === grupo)
      .filter((c) => !q || c.puesto.toLowerCase().includes(q) || c.organismo.toLowerCase().includes(q))
      .sort((a, b) => diasRestantes(a.fechaLimite) - diasRestantes(b.fechaLimite));
  }, [convocatorias, query, provincia, grupo]);

  const totalPlazas = useMemo(() => convocatorias.reduce((a, c) => a + c.plazas, 0), [convocatorias]);

  const runSync = useCallback(() => {
    if (poolIndexRef.current >= RADAR_POOL.length) {
      setToast("Radar sin novedades: la base de datos ya está al día.");
      setLastSync(new Date());
      setTimeout(() => setToast(null), 4200);
      return;
    }
    setSyncing(true);
    setTimeout(() => {
      const nueva = RADAR_POOL[poolIndexRef.current];
      poolIndexRef.current += 1;
      setConvocatorias((prev) => [nueva, ...prev]);
      setNewIds((prev) => new Set(prev).add(nueva.id));
      setLastSync(new Date());
      setSyncing(false);
      setToast(`Nueva convocatoria detectada: ${nueva.puesto} · ${nueva.organismo} (${nueva.plazas} plazas).`);
      setTimeout(() => setToast(null), 5200);
      setTimeout(() => {
        setNewIds((prev) => { const next = new Set(prev); next.delete(nueva.id); return next; });
      }, 8000);
    }, 1400);
  }, []);

  useEffect(() => {
    const interval = setInterval(() => {
      const now = new Date();
      if (now.getHours() === 16 && now.getMinutes() === 0) runSync();
    }, 60000);
    return () => clearInterval(interval);
  }, [runSync]);

  return (
    <div className="min-h-screen bg-slate-100 font-sans pb-10">
      <RadarToast toast={toast} onClose={() => setToast(null)} />
      <header className="bg-gradient-to-b from-slate-950 via-blue-950 to-blue-950 text-white">
        <div className="max-w-md mx-auto px-4 pt-5 pb-4">
          <div className="flex items-center gap-3">
            <RadarLogo size={44} />
            <div className="min-w-0">
              <h1 className="text-xl font-extrabold tracking-tight leading-none">OpoScope</h1>
              <p className="text-[11.5px] text-blue-200/80 mt-1">Radar autónomo de empleo público · Andalucía</p>
            </div>
          </div>
          <div className="mt-4 flex items-center justify-between rounded-xl bg-white/5 ring-1 ring-white/10 px-3 py-2.5">
            <div className="flex items-center gap-2 min-w-0">
              <span className="relative flex h-2.5 w-2.5 shrink-0">
                <span className={`absolute inline-flex h-full w-full rounded-full bg-emerald-400 ${syncing ? "animate-ping" : "opacity-0"}`} />
                <span className={`relative inline-flex h-2.5 w-2.5 rounded-full ${syncing ? "bg-emerald-400" : "bg-emerald-500"}`} />
              </span>
              <div className="min-w-0">
                <p className="text-[12px] font-semibold text-white truncate">{syncing ? "Sincronizando radar…" : "Radar activo"}</p>
                <p className="text-[11px] text-blue-200/70 truncate">Última sync {lastSync.toLocaleTimeString("es-ES", { hour: "2-digit", minute: "2-digit" })} · próxima 16:00 h</p>
              </div>
            </div>
            <button onClick={runSync} disabled={syncing} className="shrink-0 inline-flex items-center gap-1 rounded-lg bg-emerald-600 hover:bg-emerald-500 disabled:opacity-60 disabled:cursor-not-allowed text-white text-[11px] font-bold px-2.5 py-1.5 transition-colors">
              <RadioTower size={12} className={syncing ? "animate-pulse" : ""} />
              {syncing ? "..." : "Simular"}
            </button>
          </div>
          <div className="mt-3 grid grid-cols-3 gap-2">
            <div className="rounded-xl bg-white/5 ring-1 ring-white/10 px-2.5 py-2 text-center">
              <p className="text-base font-extrabold text-emerald-400">{convocatorias.length}</p>
              <p className="text-[10px] text-blue-200/70 leading-tight">Convocatorias</p>
            </div>
            <div className="rounded-xl bg-white/5 ring-1 ring-white/10 px-2.5 py-2 text-center">
              <p className="text-base font-extrabold text-emerald-400">{totalPlazas}</p>
              <p className="text-[10px] text-blue-200/70 leading-tight">Plazas totales</p>
            </div>
            <div className="rounded-xl bg-white/5 ring-1 ring-white/10 px-2.5 py-2 text-center">
              <p className="text-base font-extrabold text-emerald-400">8</p>
              <p className="text-[10px] text-blue-200/70 leading-tight">Provincias</p>
            </div>
          </div>
        </div>
      </header>

      <main className="max-w-md mx-auto px-4">
        <div className="sticky top-0 z-20 -mx-4 px-4 pt-3 pb-2 bg-slate-100/90 backdrop-blur">
          <div className="relative">
            <Search size={16} className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400" />
            <input value={query} onChange={(e) => setQuery(e.target.value)} placeholder="Buscar puesto u organismo…"
                   className="w-full rounded-xl bg-white ring-1 ring-slate-200 focus:ring-2 focus:ring-emerald-500 outline-none pl-9 pr-9 py-2.5 text-[13.5px] text-slate-800 placeholder:text-slate-400 shadow-sm" />
            {query && <button onClick={() => setQuery("")} aria-label="Borrar búsqueda" className="absolute right-2.5 top-1/2 -translate-y-1/2 text-slate-400 hover:text-slate-600"><X size={15} /></button>}
          </div>
          <div className="mt-2.5 flex items-center gap-1.5 overflow-x-auto no-scrollbar pb-1">
            {PROVINCIAS.map((p) => (
              <button key={p} onClick={() => setProvincia(p)}
                      className={`shrink-0 inline-flex items-center gap-1 rounded-full px-3 py-1.5 text-[12px] font-semibold ring-1 transition-colors ${provincia === p ? "bg-blue-950 text-white ring-blue-950" : "bg-white text-slate-600 ring-slate-200 hover:ring-slate-300"}`}>
                {p !== "Todas" && <MapPin size={11} />}{p}
              </button>
            ))}
          </div>
          <div className="mt-1.5 flex items-center gap-1.5 overflow-x-auto no-scrollbar pb-1">
            {GRUPOS.map((g) => (
              <button key={g} onClick={() => setGrupo(g)}
                      className={`shrink-0 inline-flex items-center gap-1 rounded-full px-3 py-1.5 text-[12px] font-semibold ring-1 transition-colors ${grupo === g ? "bg-emerald-600 text-white ring-emerald-600" : "bg-white text-slate-600 ring-slate-200 hover:ring-slate-300"}`}>
                {g !== "Todos" && <GraduationCap size={11} />}{g}
              </button>
            ))}
          </div>
        </div>

        <div className="mt-1 flex items-center justify-between px-0.5">
          <p className="text-[12px] text-slate-500">{filtered.length} {filtered.length === 1 ? "resultado" : "resultados"}</p>
          {(provincia !== "Todas" || grupo !== "Todos" || query) && (
            <button onClick={() => { setProvincia("Todas"); setGrupo("Todos"); setQuery(""); }} className="text-[12px] font-semibold text-emerald-700 hover:text-emerald-800">Limpiar filtros</button>
          )}
        </div>

        <div className="mt-2 grid grid-cols-1 gap-3">
          {filtered.map((item) => <OpoCard key={item.id} item={item} isNew={newIds.has(item.id)} />)}
          {filtered.length === 0 && (
            <div className="rounded-2xl bg-white ring-1 ring-slate-200 p-6 text-center">
              <RadioTower size={22} className="mx-auto text-slate-300" />
              <p className="mt-2 text-sm font-semibold text-slate-700">Sin señal en el radar</p>
              <p className="text-[12.5px] text-slate-500 mt-1">Ninguna convocatoria coincide con estos filtros. Prueba a ampliar la búsqueda.</p>
            </div>
          )}
        </div>

        <footer className="mt-8 text-center">
          <div className="inline-flex items-center gap-1.5 text-[11px] text-slate-400">
            <CheckCircle2 size={12} className="text-emerald-500" />
            Fuentes: BOJA, BOE y Boletines Oficiales de las ocho provincias andaluzas
          </div>
          <p className="text-[10.5px] text-slate-300 mt-1">OpoScope sincroniza automáticamente cada día a las 16:00 h</p>
        </footer>
      </main>
    </div>
  );
}
